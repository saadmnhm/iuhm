    .then(res => res.json())
    .then(data => console.log(data.data));

  React
  ─────
  const [news, setNews] = useState([]);
  useEffect(() => {
    fetch('https://domain/api/v1/news/latest?limit=5')
      .then(r => r.json())
      .then(d => setNews(d.data));
  }, []);

  );

  import requests
